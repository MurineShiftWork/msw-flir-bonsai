    class BonsaiCameraRunner:
    """Manage a single Bonsai camera acquisition subprocess.

    Args:
        bonsai_exe: Path to ``Bonsai.exe``. Falls back to ``BONSAI_EXE`` env var.
        workflow: Workflow stem name (without ``.bonsai``),
            e.g. ``"run-flir-flycap-1cam"``.
        acqdir: Full path to the pre-created video_flir acquisition directory.
        cam_basename: Pre-computed file stem for this camera, e.g.
            ``"mouse01__dt__video_flir.mswflir.top.0"``.  The workflow appends
            ``.avi`` / ``.timestamps.csv`` to produce output filenames.
        cam_index: Camera SDK enumeration index (``cam1idx`` workflow property).
        cam_serial: Camera serial number (``cam1serial``, Spinnaker only). When
            set, the Spinnaker node selects the camera by serial and ignores
            the index; empty keeps index selection.
        fps: Target frame rate (FlyCapture only; ignored for Spinnaker).  Frame
            rate should be configured in camera non-volatile memory; only pass
            this for an explicit CLI override.
        driver: ``"flycap"`` or ``"spinnaker"``: controls which properties are passed.
        extra_props: Additional ``-p key=value`` pairs forwarded to Bonsai CLI.
        startup_timeout: Seconds to wait for the subprocess to start before raising.
        start_timeout: Seconds after launch for the first frame to reach the
            ``.avi`` (:meth:`check_started`).
        retry_delay: Seconds to wait before the single relaunch after Bonsai
            exited without a frame (:meth:`check_started`); None disables it.
    """

    def __init__(
        self,
        workflow: str,
        acqdir: str | Path,
        cam_basename: str,
        cam_index: int = 0,
        fps: int | None = None,
        driver: str = "flycap",
        cam_serial: str = "",
        bonsai_exe: str | Path | None = None,
        extra_props: dict[str, str] | None = None,
        startup_timeout: float = _STARTUP_TIMEOUT,
        start_timeout: float = _START_TIMEOUT_S,
        retry_delay: float | None = _EARLY_EXIT_RETRY_DELAY_S,
    ) -> None:
        self._bonsai_exe = _find_bonsai_exe(bonsai_exe)
        self._workflow_path = workflow_path(workflow)
        self._acqdir = str(acqdir)
        self._cam_basename = cam_basename
        self._cam_index = cam_index
        self._cam_serial = cam_serial
        self._fps = fps
        self._driver = driver
        self._extra_props = extra_props or {}
        self._startup_timeout = startup_timeout
        self._start_timeout = start_timeout
        self._avi_before: tuple[int, int] | None = None
        self._retry_delay = retry_delay
        self._exited_unexpectedly = False
        # The monitor stays quiet while check_started runs and about an exit
        # the check has already logged, so one failure is logged exactly once.
        self._checking = False
        self._exit_reported: subprocess.Popen | None = None

        self._process: subprocess.Popen | None = None
        self._monitor_thread: threading.Thread | None = None
        self._stopped = threading.Event()
        self._stop_file: Path | None = None
        self._layout_file: Path | None = None
        self._started_at = 0.0

    def _build_cmd(self) -> list[str]:
        """Assemble the Bonsai subprocess command list with all -p property flags."""
        props: dict[str, str] = {
            "acqdir": f'"{self._acqdir}"',
            "cam_basename": f'"{self._cam_basename}"',
            "cam1idx": str(self._cam_index),
        }
        if self._driver == "flycap":
            if self._fps is not None:
                props["cam1fps"] = str(self._fps)
            props["cam1ts"] = "True"
            props["cam1framecounter"] = "True"
        elif self._driver == "spinnaker" and self._cam_serial:
            props["cam1serial"] = self._cam_serial

        props.update(self._extra_props)

        cmd = [
            self._bonsai_exe,
            str(self._workflow_path),
            "--start",
            "--no-editor",
        ]
        layout = self._preview_layout()
        if layout is not None:
            cmd += ["--visualizer-layout", str(layout)]
        for key, value in props.items():
            cmd += ["-p", f"{key}={value}"]
        return cmd

    def _preview_layout(self) -> Path | None:
        """Per-camera copy of the workflow's ``.bonsai.layout`` (live preview window).

        The window is shifted right by camera index so several cameras sit side by
        side. Returns None when the workflow has no layout file.
        """
        template = Path(f"{self._workflow_path}.layout")
        if not template.exists():
            return None
        x = 40 + self._cam_index * 660
        xml = template.read_text(encoding="utf-8").replace(
            "<Location><X>40</X>", f"<Location><X>{x}</X>", 1
        )
        out = (
            Path(tempfile.gettempdir())
            / f"msw-bonsai-{os.getpid()}-cam{self._cam_index}.layout"
        )
        out.write_text(xml, encoding="utf-8")
        self._layout_file = out
        return out

    def start(self) -> None:
        """Launch the Bonsai subprocess and begin background monitoring."""
        if self._process is not None and self._process.poll() is None:
            raise RuntimeError("BonsaiCameraRunner is already running.")

        cmd = self._build_cmd()
        logging.info(f"[BonsaiCamera cam{self._cam_index}] Starting: {' '.join(cmd)}")
        self._write_metadata()

        self._stopped.clear()
        self._exited_unexpectedly = False
        self._avi_before = self._avi_stat()
        # Graceful stop: Extensions/key-ctrl-stop.bonsai polls this file and
        # completes the stream once it exists, so the AVI/CSV writers flush and
        # close and Bonsai exits.
        self._stop_file = (
            Path(tempfile.gettempdir())
            / f"msw-bonsai-{os.getpid()}-cam{self._cam_index}.stop"
        )
        self._stop_file.unlink(missing_ok=True)
        self._process = subprocess.Popen(
            cmd,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            env={
                **os.environ,
                "MSW_BONSAI_STOP_FILE": str(self._stop_file),
                "MSW_BONSAI_PARENT_PID": str(os.getpid()),
            },
            # Own process group: a Ctrl+C in the caller's console must not kill
            # Bonsai before the caller has shut down and requested a clean stop.
            creationflags=getattr(subprocess, "CREATE_NEW_PROCESS_GROUP", 0),
        )

        self._started_at = time.monotonic()
        self._monitor_thread = threading.Thread(
            target=self._monitor,
            args=(self._process,),
            daemon=True,
            name=f"bonsai-cam{self._cam_index}-monitor",
        )
        self._monitor_thread.start()
        logging.info(
            f"[BonsaiCamera cam{self._cam_index}] Process started "
            f"(pid={self._process.pid})"
        )

    def _write_metadata(self) -> None:
        """Write the per-camera metadata file (driver, CSV layout); never raises."""
        from msw_flir_bonsai.metadata import camera_metadata, write_camera_metadata

        meta = camera_metadata(
            driver=self._driver,
            workflow=self._workflow_path.stem,
            cam_index=self._cam_index,
            fps=self._fps,
        )
        write_camera_metadata(self._acqdir, self._cam_basename, meta)

    @property
    def avi_path(self) -> Path:
        """The video file the workflow writes: ``<acqdir>/<cam_basename>.avi``."""
        return Path(self._acqdir) / f"{self._cam_basename}.avi"

    def _avi_stat(self) -> tuple[int, int] | None:
        try:
            st = self.avi_path.stat()
        except OSError:
            return None
        return st.st_size, st.st_mtime_ns

    def _start_status(self) -> str | None:
        """``""`` once the first frame is written, a failure reason, or None."""
        stat = self._avi_stat()
        # a file left from an earlier run (same size and mtime) is not a frame
        if stat is not None and stat[0] > 0 and stat != self._avi_before:
            return ""
        assert self._process is not None
        rc = self._process.poll()
        if rc is not None:
            self._exit_reported = self._process  # check_started logs it
            age = time.monotonic() - self._started_at
            return (
                f"Bonsai cam{self._cam_index} exited {age:.1f} s after launch "
                f"without a frame (returncode={rc}): camera missing or in use, "
                "or the workflow failed"
            )
        return None

    def check_started(
        self,
        timeout: float | None = None,
        expect_present: bool = True,
        deadline: float | None = None,
    ) -> None:
        """Block until the first frame is in the ``.avi``; raise if it never is.

        Polls every 0.25 s. The start fails when Bonsai exits first (at any
        time; its return code is 0 either way) or when ``timeout`` (default
        ``start_timeout``) passes with Bonsai alive but no frame; Bonsai is
        then stopped. After an exit without a frame Bonsai is relaunched once
        after ``retry_delay`` s, because the camera may not be released yet,
        unless ``expect_present=False`` (the caller knows the camera is
        missing, so the relaunch would fail too). The relaunch gets a fresh
        ``timeout``.

        Args:
            timeout: Seconds from launch to wait for the first frame.
            expect_present: False skips the relaunch.
            deadline: ``time.monotonic()`` deadline for the first attempt,
                shared by several cameras; overrides ``timeout`` for it.

        Raises:
            BonsaiStartError: No frame: Bonsai exited (on the relaunch too)
                or the timeout passed.
        """
        if self._process is None:
            return
        self._checking = True
        try:
            self._check_started(timeout, expect_present, deadline)
        finally:
            self._checking = False

    def _check_started(
        self, timeout: float | None, expect_present: bool, deadline: float | None
    ) -> None:
        if timeout is None:
            timeout = self._start_timeout
        if deadline is None:
            deadline = self._started_at + timeout
        msg = self._wait_first_frame(deadline)
        if msg is None:
            return
        exited = msg.startswith("Bonsai")  # a timeout is never relaunched
        if exited and not expect_present:
            msg += " (no relaunch: camera not detected)"
        if not exited or not expect_present or self._retry_delay is None:
            logging.error(f"[BonsaiCamera cam{self._cam_index}] {msg}")
            raise BonsaiStartError(msg)
        logging.warning(
            f"[BonsaiCamera cam{self._cam_index}] {msg}; attempt 1 of 2, "
            f"relaunching in {self._retry_delay:g} s"
        )
        time.sleep(self._retry_delay)
        self.start()
        msg = self._wait_first_frame(self._started_at + timeout)
        if msg is None:
            logging.info(f"[BonsaiCamera cam{self._cam_index}] Relaunch recording")
            return
        logging.error(f"[BonsaiCamera cam{self._cam_index}] {msg}; attempt 2 of 2")
        raise BonsaiStartError(f"{msg} (after one relaunch)")

    def _wait_first_frame(self, deadline: float) -> str | None:
        """None once a frame is written, else the failure reason.

        Exit reasons start with ``"Bonsai"``; on a timeout Bonsai is stopped
        and the reason starts with ``"no frame"``.
        """
        while True:
            status = self._start_status()
            if status == "":
                return None
            if status is not None:
                return status
            if time.monotonic() >= deadline:
                self.stop()
                return (
                    f"no frame from cam{self._cam_index} in {self.avi_path.name} "
                    "before the start timeout; Bonsai stopped"
                )
            time.sleep(_START_POLL_S)

    def _monitor(self, proc: subprocess.Popen) -> None:
        """Poll ``proc`` in a daemon thread and record an exit before stop.

        Takes the process it watches: after a relaunch the old thread must not
        flag the new process. While check_started runs, and for an exit it
        has logged, the monitor only flags the exit; otherwise it logs it.
        """
        while not self._stopped.is_set() and proc is self._process:
            rc = proc.poll()
            if rc is not None:
                if not self._stopped.is_set() and proc is self._process:
                    self._exited_unexpectedly = True
                    if not self._checking and proc is not self._exit_reported:
                        logging.error(
                            f"[BonsaiCamera cam{self._cam_index}] "
                            f"Process exited unexpectedly (returncode={rc})"
                        )
                break
            time.sleep(0.5)

    def stop(self, timeout: float = 5.0) -> None:
        """Gracefully terminate the Bonsai subprocess."""
        if self._process is None:
            return
        self._stopped.set()
        if self._process.poll() is None:
            logging.info(f"[BonsaiCamera cam{self._cam_index}] Requesting stop")
            if self._stop_file is not None:
                self._stop_file.touch()
            try:
                self._process.wait(timeout=timeout)
            except subprocess.TimeoutExpired:
                logging.warning(
                    f"[BonsaiCamera cam{self._cam_index}] Stop timed out; killing"
                )
                if os.name == "nt":
                    # Bonsai.exe relaunches itself as a child worker; kill() would only
                    # kill the launcher and orphan the worker. Kill the whole tree.
                    subprocess.run(
                        ["taskkill", "/F", "/T", "/PID", str(self._process.pid)],
                        capture_output=True,
                    )
                else:
                    self._process.kill()
                self._process.wait(timeout=5.0)
        for f in (self._stop_file, self._layout_file):
            if f is not None:
                f.unlink(missing_ok=True)

    def wait(self, timeout: float | None = None) -> int | None:
        """Block until the subprocess exits. Returns the return code."""
        if self._process is None:
            return None
        try:
            return self._process.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            return None

    @property
    def exited_unexpectedly(self) -> bool:
        """True if Bonsai exited before :meth:`stop` was requested.

        Bonsai exits 0 when it cannot record, so this, not the return code,
        tells a caller that the recording failed. Check it before :meth:`stop`.
        """
        if self._exited_unexpectedly:
            return True
        return (
            self._process is not None
            and not self._stopped.is_set()
            and self._process.poll() is not None
        )

    @property
    def is_running(self) -> bool:
        """True if the subprocess exists and has not yet exited."""
        return self._process is not None and self._process.poll() is None

    @property
    def pid(self) -> int | None:
        """OS process ID of the Bonsai subprocess, or None if not started."""
        return self._process.pid if self._process else None

                workflow=resolved_workflow,
                acqdir=acqdir,
                cam_basename=cam_basenames[i],
                cam_index=i,
                fps=fps,
                driver=driver,
                bonsai_exe=bonsai_exe,
            )
            for i in range(len(cam_basenames))
        ]
        return cls(runners)

    def start(self, timeout: float | None = None, expect_present: bool = True) -> None:
        """Start all camera subprocesses, then wait for every first frame.

        The cameras are waited for in parallel against one shared deadline
        (``timeout`` from now, default each runner's ``start_timeout``).
        ``timeout=0`` skips the wait. ``expect_present`` is passed to
        :meth:`BonsaiCameraRunner.check_started`.

        Raises:
            BonsaiStartError: A camera wrote no frame; all cameras are stopped
                before raising.
        """
        for runner in self._runners:
            runner.start()
        if timeout is not None and timeout <= 0:
            return
        deadline = None if timeout is None else time.monotonic() + timeout
        failed: list[str] = []

        def wait(runner: BonsaiCameraRunner) -> None:
            try:
                runner.check_started(
                    timeout, expect_present=expect_present, deadline=deadline
                )
            except BonsaiStartError as e:
                failed.append(str(e))

        threads = [
            threading.Thread(target=wait, args=(r,), daemon=True) for r in self._runners
        ]
        for t in threads:
            t.start()
        for t in threads:
            t.join()
        if failed:
            self.stop()
            raise BonsaiStartError("; ".join(failed))

    def stop(self, timeout: float = 5.0) -> None:
        """Stop all camera subprocesses in parallel and wait for them to exit."""
        threads = []
        for runner in self._runners:
            t = threading.Thread(
                target=runner.stop, kwargs={"timeout": timeout}, daemon=True
            )
            t.start()
            threads.append(t)
        for t in threads:
            t.join(timeout=timeout + 2)

    def all_running(self) -> bool:
        """True if every camera subprocess is still active."""
        return all(r.is_running for r in self._runners)

    @property
    def exited_unexpectedly(self) -> bool:
        """True if any camera's Bonsai exited before stop was requested."""
        return any(r.exited_unexpectedly for r in self._runners)

    @property
    def any_running(self) -> bool:
        """True if at least one camera subprocess is still active."""
        return any(r.is_running for r in self._runners)

    units: Sequence[Any], driver: str, probe: ProbeResult
) -> list[CameraLaunch]:
    """Turn configured cameras and a probe into the launch plan.

    ``units`` are objects with ``index``, ``name`` and an optional ``serial``
    (msw-core ``CameraUnit``). The plan keeps their order.

    - Source ``"none"``: warn and launch by configured index (Spinnaker still
      gets configured serials).

      collide with a serial-resolved one.

      serial connected. USB gives no index order, so FlyCapture keeps the
      configured index (warning).

    Raises:
        CameraMismatchError: Cameras are missing, a serial is not connected,
            or two cameras resolve to the same index.
    """

        CameraLaunch(
            name=getattr(u, "name", "") or "",
            index=int(u.index),
            serial=_norm(getattr(u, "serial", "")),
        )
        for u in units
    ]

        return plan

        log.warning(
            f"Camera probe unavailable ({probe.error or 'no source'}): "
            "launching by configured index"
        )
        return _finish(plan, pass_serial)

    count = probe.count or 0
    label = f"{count} {driver} camera(s) found via {probe.source}"
    if not any(c.serial for c in plan):
        needed = max(c.index for c in plan) + 1
        if count < needed:
            raise CameraMismatchError(
                f"setup needs {needed} camera(s) (highest index {needed - 1}); {label}"
            )
        return _finish(plan, pass_serial)

    if probe.source == "usb":
        if count < len(plan):
            raise CameraMismatchError(f"setup needs {len(plan)} camera(s); {label}")
        found_usb = {serial_key(c.serial) for c in probe.cameras}
        missing = [
            c.serial for c in plan if c.serial and serial_key(c.serial) not in found_usb
        ]
        if missing:
            raise CameraMismatchError(
                f"camera serial(s) {missing} not connected; {label} (source=usb): "
                f"{sorted(found_usb)}"
            )
        if not pass_serial:
            log.warning("No SDK index order via USB: FlyCapture uses configured index")
        return _finish(plan, pass_serial)

    by_serial = {
        serial_key(c.serial): c.index
        for c in probe.cameras
        if c.serial and c.index is not None
    }
    missing = [
        c.serial for c in plan if c.serial and serial_key(c.serial) not in by_serial
    ]
    if missing:
        raise CameraMismatchError(
            f"camera serial(s) {missing} not connected; {label}: {sorted(by_serial)}"
        )
    for c in plan:
        if c.serial:
            c.index = by_serial[serial_key(c.serial)]
        elif c.index >= count:
            raise CameraMismatchError(
                f"camera {c.name or c.index!r} has no serial and index {c.index} "
                f"is not connected; {label}"
            )
    indices = [c.index for c in plan]
    if len(set(indices)) != len(indices):
        raise CameraMismatchError(f"two cameras resolve to one index: {indices}")
    return _finish(plan, pass_serial)

    class FlirBonsaiBackend:
    """FLIR cameras recorded by one Bonsai process each.

    Args:
        probe: ``probe(driver) -> ProbeResult``; defaults to
            :func:`probe_cameras` with a 10 s timeout.
        runner_factory: Builds one runner per camera with the
            :class:`BonsaiCameraRunner` keyword arguments; defaults to that
            class.
    """

    acq_type = "video_flir"
    version = __version__

    def __init__(
        self,
        *,
        probe: Callable[[str], ProbeResult] | None = None,
        runner_factory: Callable[..., Any] | None = None,
    ) -> None:
        self._probe = probe or (lambda d: probe_cameras(d, timeout=_PROBE_TIMEOUT_S))
        self._runner_factory = runner_factory or BonsaiCameraRunner
        self._settings = FlirSettings()
        self._bonsai_exe = ""
        self._units: list[FlirUnit] = []
        self._plan: dict[str, CameraLaunch] = {}
        self._serials: dict[str, str] = {}
        self._preflight: dict[str, UnitResult] | None = None
        self._runners: dict[str, Any] = {}
        self._files: dict[str, list[str]] = {}
        self._results: dict[str, UnitResult] = {}
        self._stopped: dict[str, UnitResult] | None = None
        self._acqdir = ""

    # ------------------------------------------------------------------
    # configure / preflight

    def configure(
        self, settings: Mapping[str, Any], units: list[CameraUnitSpec]
    ) -> None:
        """Validate and store the settings and cameras.

        Raises:
            CameraConfigError: Unknown or invalid settings, invalid cameras, or
                no ``Bonsai.exe`` (:func:`resolve_bonsai_exe`).
        """
        self._settings = parse_settings(settings)
        self._units = parse_units(list(units), self._settings.n_cameras)
        self._bonsai_exe = resolve_bonsai_exe(self._settings.bonsai_exe)
        self._plan = {}
        self._serials = {}
        self._preflight = None
        self._results = {}
        self._stopped = None

    def preflight(self) -> list[UnitResult]:
        """Probe the cameras and resolve each one's launch index.

        Returns:
            Per camera ``ready`` (``extra``: probe source, resolved index,
            serial, and ``probe_serials`` when the probe has no index order)
            or ``missing`` (the probe message as ``error``); every
            camera ``failed`` (``bonsai_not_found``) when ``Bonsai.exe`` is gone.
        """
        driver = self._settings.driver
        if not Path(self._bonsai_exe).is_file():
            error = f"Bonsai.exe not found: {self._bonsai_exe!r}"
            log.error(f"flir_bonsai: {error}")
            self._plan = {}
            self._serials = {}
            gone = {
                u.name: UnitResult(
                    name=u.name,
                    ok=False,
                    state=UNIT_FAILED,
                    error=error,
                    error_code="bonsai_not_found",
                )
                for u in self._units
            }
            self._preflight = gone
            self._results = dict(gone)
            return self._ordered(gone)
        try:
            probe = self._probe(driver)
        except Exception as e:  # noqa: BLE001  the probe must not break a run
            probe = ProbeResult(driver, error=f"probe raised {e!r}")
        source = probe.source if probe.source != "none" else "unavailable"
        at_index = {
            c.index: c.serial for c in probe.cameras if c.index is not None and c.serial
        }
        unordered = [c.serial for c in probe.cameras if c.index is None and c.serial]
        unordered_keys = {serial_key(s) for s in unordered}
        self._plan = {}
        self._serials = {}
        results: dict[str, UnitResult] = {}
        for unit, launch, error in self._resolve(probe):
            extra: dict[str, Any] = {"probe": source}
            serial = unit.serial
            if probe.source != "none":
                if launch is not None and launch.index in at_index:
                    serial = at_index[launch.index]
                elif unordered:
                    if serial_key(unit.serial) not in unordered_keys:
                        serial = ""
                    extra["probe_serials"] = list(unordered)
            if serial:
                extra["serial"] = serial
                self._serials[unit.name] = serial
            if launch is None:
                results[unit.name] = UnitResult(
                    name=unit.name,
                    ok=False,
                    state=UNIT_MISSING,
                    error=error,
                    error_code="not_connected",
                    extra=extra,
                )
                continue
            self._plan[unit.name] = launch
            extra["index"] = launch.index
            results[unit.name] = UnitResult(
                name=unit.name, ok=True, state=UNIT_READY, extra=extra
            )
        self._preflight = results
        self._results = dict(results)
        return self._ordered(results)

    def _resolve(
        self, probe: ProbeResult
    ) -> list[tuple[FlirUnit, CameraLaunch | None, str]]:
        """Launch plan per unit, or None with the reason it is missing."""
        driver = self._settings.driver

        def probe_units(units: list[FlirUnit]) -> list[_ProbeUnit]:
            return [_ProbeUnit(u.name, u.index, u.serial) for u in units]

        try:
            plan = resolve_camera_units(probe_units(self._units), driver, probe)
            return [(u, p, "") for u, p in zip(self._units, plan, strict=True)]
        except CameraMismatchError as e:
            group_error = str(e)
        missing: dict[str, str] = {}
        for u in self._units:
            try:
                resolve_camera_units(probe_units([u]), driver, probe)
            except CameraMismatchError as e:
                missing[u.name] = str(e)
        rest = [u for u in self._units if u.name not in missing]
        resolved: dict[str, CameraLaunch] = {}
        if rest:
            try:
                plan = resolve_camera_units(probe_units(rest), driver, probe)
                resolved = {p.name: p for p in plan}
            except CameraMismatchError as e:
                missing.update({u.name: str(e) for u in rest})
        out: list[tuple[FlirUnit, CameraLaunch | None, str]] = []
        for u in self._units:
            if u.name in resolved:
                out.append((u, resolved[u.name], ""))
            else:
                out.append((u, None, missing.get(u.name, group_error)))
        return out

    # ------------------------------------------------------------------
    # start

    def start(self, target: CameraTarget) -> list[UnitResult]:
        """Launch one Bonsai per ready camera and wait for each first frame.

        Runs :meth:`preflight` first if it has not run. Cameras that failed
        preflight are not launched and keep their preflight result.

        Returns:
            Per camera ``recording``, ``failed`` (no first frame, or the launch
            itself failed) or ``missing``.

        Raises:
            RuntimeError: Called again while cameras are recording.
        """
        if self._runners:
            raise RuntimeError("flir_bonsai: start called while recording")
        if self._preflight is None:
            self.preflight()
        assert self._preflight is not None
        self._acqdir = target.acqdir
        self._stopped = None
        self._files = {}
        results: dict[str, UnitResult] = {
            name: r for name, r in self._preflight.items() if not r.ok
        }
        s = self._settings
        to_check: dict[str, Any] = {}
        for unit in self._units:
            launch = self._plan.get(unit.name)
            if launch is None:
                continue
            stem = camera_stem(target.basename, unit.label)
            self._files[unit.name] = [f"{stem}.avi", f"{stem}.timestamps.csv"]
            extra = self._extra(unit, launch)
            try:
                runner = self._runner_factory(
                    workflow=s.workflow,
                    acqdir=target.acqdir,
                    cam_basename=stem,
                    cam_index=launch.index,
                    driver=s.driver,
                    cam_serial=launch.serial,
                    bonsai_exe=self._bonsai_exe,
                )
                runner.start()
            except Exception as e:  # noqa: BLE001  one camera must not stop the rest
                log.error(f"flir_bonsai: {unit.name} could not launch: {e}")
                results[unit.name] = UnitResult(
                    name=unit.name,
                    ok=False,
                    state=UNIT_FAILED,
                    error=f"launch failed: {e}",
                    error_code="launch_failed",
                    extra=extra,
                )
                continue
            self._runners[unit.name] = runner
            to_check[unit.name] = runner

        errors: dict[str, str] = {}

        def check(name: str, runner: Any) -> None:
            try:
                runner.check_started()
            except BonsaiStartError as e:
                errors[name] = str(e)
            except Exception as e:  # noqa: BLE001
                errors[name] = f"start check failed: {e!r}"

        threads = [
            threading.Thread(target=check, args=item, daemon=True)
            for item in to_check.items()
        ]
        for t in threads:
            t.start()
        for t in threads:
            t.join()

        for unit in self._units:
            runner = to_check.get(unit.name)
            if runner is None:
                continue
            extra = self._extra(unit, self._plan[unit.name])
            if unit.name in errors:
                self._runners.pop(unit.name, None)
                _safe_stop(runner, _STOP_TIMEOUT_S)
                results[unit.name] = UnitResult(
                    name=unit.name,
                    ok=False,
                    state=UNIT_FAILED,
                    error=errors[unit.name],
                    error_code="no_first_frame",
                    files=self._existing_files(unit.name),
                    extra=extra,
                )
            else:
                results[unit.name] = UnitResult(
                    name=unit.name,
                    ok=True,
                    state=UNIT_RECORDING,
                    files=list(self._files[unit.name]),
                    extra={**extra, "pid": getattr(runner, "pid", None)},
                )
        self._results = dict(results)
        return self._ordered(results)

    # ------------------------------------------------------------------
    # stop / close

    def stop(self, reason: str) -> list[UnitResult]:
        """Stop every recording camera and judge what it saved.

        A camera is ``stopped`` when Bonsai exited after the stop request and
        its timestamps file has data, else ``failed``: it did not exit, it
        exited before the stop (Bonsai exits 0 without recording), or the
        timestamps file is missing or empty. ``extra["returncode"]`` holds the
        exit code. Cameras that never recorded keep their earlier result.
        Calling it again returns the same results.
        """
        if self._stopped is not None:
            return self._ordered(self._stopped)
        log.info(f"flir_bonsai: stopping {sorted(self._runners)} ({reason})")
        runners = dict(self._runners)
        early = {name: bool(r.exited_unexpectedly) for name, r in runners.items()}
        threads = [
            threading.Thread(target=_safe_stop, args=(r, _STOP_TIMEOUT_S), daemon=True)
            for r in runners.values()
        ]
        for t in threads:
            t.start()
        for t in threads:
            t.join(timeout=_STOP_TIMEOUT_S + 2)

        results = dict(self._results)
        for unit in self._units:
            runner = runners.get(unit.name)
            if runner is None:
                prev = results.get(unit.name)
                if prev is None or prev.state in (UNIT_READY, UNIT_RECORDING):
                    results[unit.name] = UnitResult(
                        name=unit.name, ok=True, state=UNIT_STOPPED
                    )
                continue
            rc = runner.wait(timeout=_EXIT_WAIT_S)
            extra = {**self._extra(unit, self._plan[unit.name]), "returncode": rc}
            files = self._existing_files(unit.name)
            error, code = "", ""
            if rc is None:
                error, code = "Bonsai did not exit after the stop request", "no_exit"
            elif early[unit.name]:
                error = f"Bonsai exited before the stop (returncode={rc})"
                code = "exited_early"
            elif not self._timestamps_ok(unit.name):
                error = "timestamps file missing or empty"
                code = "empty_timestamps"
            results[unit.name] = UnitResult(
                name=unit.name,
                ok=not error,
                state=UNIT_FAILED if error else UNIT_STOPPED,
                error=error,
                error_code=code,
                files=files,
                extra=extra,
            )
            if error:
                log.error(f"flir_bonsai: {unit.name}: {error}")
        self._runners = {
            n: r for n, r in runners.items() if getattr(r, "is_running", False)
        }
        self._results = dict(results)
        self._stopped = results
        return self._ordered(results)

    def close(self) -> None:
        """Kill any Bonsai still running. Safe to call repeatedly."""
        for name, runner in list(self._runners.items()):
            if getattr(runner, "is_running", False):
                log.warning(f"flir_bonsai: killing {name} on close")
                _safe_stop(runner, 0.0)
        self._runners = {}

    # ------------------------------------------------------------------
    # helpers

    def _ordered(self, results: dict[str, UnitResult]) -> list[UnitResult]:
        return [results[u.name] for u in self._units if u.name in results]

    def _extra(self, unit: FlirUnit, launch: CameraLaunch) -> dict[str, Any]:
        extra: dict[str, Any] = {"index": launch.index}
        serial = self._serials.get(unit.name, unit.serial)
        if serial:
            extra["serial"] = serial
        return extra

    def _existing_files(self, name: str) -> list[str]:
        root = Path(self._acqdir)
        return [f for f in self._files.get(name, []) if (root / f).is_file()]

    def _timestamps_ok(self, name: str) -> bool:
        ts = Path(self._acqdir) / self._files[name][1]
        try:
            return ts.stat().st_size > 0
        except OSError:
            return False

    def align_ttl_edges(
    cam_df: pd.DataFrame,
    bpod_event_times_s: Sequence[float],
    gpio_col: str = "gpio_state",
    max_offset_search_s: float = 60.0,
    min_matches: int = 5,
    line_bit: int = FLYCAP_BARCODE_LINE_BIT,
) -> float:
    """Estimate camera-to-Bpod offset by matching TTL rising edges.

    The sequence task and others pulse a Bpod BNC line at trial start/end.
    The camera records these as GPIO state changes.  This function cross-correlates
    the two edge sequences to find the best time offset.

    Args:
        cam_df: Preprocessed camera DataFrame.
        bpod_event_times_s: Trial event times (e.g. trial-start) in Bpod clock.
        gpio_col: Camera GPIO column name (line status bitmask).
        max_offset_search_s: Maximum |offset| to consider during search.
        min_matches: Minimum number of matching edges required.
        line_bit: Bit of the TTL line in *gpio_col* (31 on FlyCapture).

    Returns:
        ``offset_s`` such that ``bpod_time = camera_time + offset_s``.
    """
    gpio = line_state(cam_df[gpio_col].to_numpy(), line_bit)
    ts = cam_df["timestamp_s"].to_numpy()
    rising_idx = np.where((gpio[1:] == 1) & (gpio[:-1] == 0))[0] + 1
    cam_edge_times = ts[rising_idx]

    bpod_arr = np.asarray(bpod_event_times_s)

    # Search over candidate offsets = differences between all pairs of edges
    candidates = []
    for ce in cam_edge_times:
        for be in bpod_arr:
            candidates.append(be - ce)

    candidates_arr = np.asarray(candidates)
    candidates_arr = candidates_arr[np.abs(candidates_arr) <= max_offset_search_s]

    if len(candidates_arr) == 0:
        raise ValueError(
            "No candidate offsets within search range: check GPIO and event times."
        )

    # Histogram to find consensus offset
    hist, edges = np.histogram(candidates_arr, bins=int(max_offset_search_s * 20))
    best_bin = int(np.argmax(hist))
    offset = float((edges[best_bin] + edges[best_bin + 1]) / 2)

    matches = int(hist[best_bin])
    if matches < min_matches:
        raise ValueError(
            f"TTL alignment found only {matches} matching edges (need {min_matches}). "
            "Consider increasing max_offset_search_s or checking GPIO."
        )

    logging.info(f"TTL-edge alignment: {matches} matching edges, offset={offset:.4f}s")
    return offset

    def workflow_path(name: str) -> Path:
    """Return absolute path to a bundled .bonsai workflow file by stem name."""
    p = WORKFLOW_DIR / f"{name}.bonsai"
    if not p.exists():
        available = sorted(f.stem for f in WORKFLOW_DIR.glob("*.bonsai"))
        raise FileNotFoundError(f"Workflow '{name}' not found. Available: {available}")
    return p
